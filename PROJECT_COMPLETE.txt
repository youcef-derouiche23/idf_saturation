     -e POSTGRES_PASSWORD=idfm_pass \
     -e POSTGRES_USER=idfm_user \
     -e POSTGRES_DB=idfm_datamarts \
     -p 5433:5432 -d postgres:15

2️⃣ LANCER LE PIPELINE (2-5 minutes)

   bash /Users/youcef/Downloads/Projet_IDFM_Frequentation/start_full_pipeline.sh

3️⃣ LANCER L'API (Terminal 1)

   cd /Users/youcef/Downloads/Projet_IDFM_Frequentation/api
   python -m uvicorn app:app --reload --port 8000

4️⃣ LANCER LE DASHBOARD (Terminal 2)

   cd /Users/youcef/Downloads/Projet_IDFM_Frequentation
   streamlit run dashboard/app.py

5️⃣ ACCÉDER AUX SERVICES

   📍 API Swagger : http://localhost:8000/docs
   📍 Dashboard  : http://localhost:8501

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

DM1: dm_frequentation_par_station_ligne
    Colonnes: ligne, id_station, nom_station, heure, jour_semaine,
              nb_validations, rank_station_par_ligne
    Cas d'usage: Top stations par ligne

DM2: dm_regularite_par_ligne
    Colonnes: date, ligne, nom_ligne, taux_ponctualite, nb_retards,
              delai_moyen, rang_regularite
    Cas d'usage: Comparer régularité inter-lignes

DM3: dm_evolution_frequentation
    Colonnes: date, jour_semaine, periode_vacances, ligne, id_station,
              nom_station, nb_validations_cumul, evolution_vs_semaine_precedente
    Cas d'usage: Tendances temporelles

DM4: dm_saturation_ml ⭐ ML-READY
    Colonnes: date, heure, ligne, id_station, nom_station, nb_validations,
              taux_ponctualite, jour_semaine, is_vacances, jour_ferie,
              est_saturation (LABEL)
    Cas d'usage: Prédire saturations futures
